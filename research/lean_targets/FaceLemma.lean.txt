open Classical

variable {n : Nat}

/-- Size of the class of position `i`. -/
noncomputable def classSize (p : ColorPattern n) (i : Fin n) : Nat :=
  Fintype.card {j : Fin n // p i j}

/-- Rigid positions: their class has size 2 or 3. -/
def IsRigid (p : ColorPattern n) (i : Fin n) : Prop :=
  classSize p i = 2 ∨ classSize p i = 3

/-- Heavy positions: their class has size at least 4. -/
def IsHeavy (p : ColorPattern n) (i : Fin n) : Prop := 4 ≤ classSize p i

/-- A rigid class separates `x < y` if it has a point strictly between them
and a point outside the closed interval. -/
def RigidSeparates (p : ColorPattern n) (x y : Fin n) : Prop :=
  ∃ c d : Fin n, IsRigid p c ∧ p c d ∧ x < c ∧ c < y ∧ (d < x ∨ y < d)

/-- Same face: no rigid class separates the two positions (order-symmetrised). -/
def SameFace (p : ColorPattern n) (x y : Fin n) : Prop :=
  ¬ RigidSeparates p (min x y) (max x y)

/-- The rigid classes are pairwise noncrossing. -/
def RigidNoncrossing (p : ColorPattern n) : Prop :=
  ∀ a b c d : Fin n, a < b → b < c → c < d → IsRigid p a → IsRigid p b →
    p a c → p b d → p a b

/-- Restriction of a pattern to an ordered subset of positions. -/
noncomputable def restrictPattern (p : ColorPattern n) (s : Finset (Fin n)) :
    ColorPattern s.card :=
  Setoid.comap (s.orderEmbOfFin rfl) p

/-- The heavy positions in the face of a heavy position `x`. -/
noncomputable def faceOf (p : ColorPattern n) (x : Fin n) : Finset (Fin n) :=
  Finset.univ.filter (fun y => IsHeavy p y ∧ SameFace p x y)
